// =================================== // MINI タスク → Google Calendar 自動登録 // =================================== function doPost(e) { try { var data = JSON.parse(e.postData.contents); var title = data.title || "（無題）"; var date = data.date || ""; var time = data.time || ""; var priority = data.priority || "中"; var duration = data.duration || 60; // 分 var calendar = CalendarApp.getDefaultCalendar(); var event; if (date && time) { // 日時あり → 時刻指定イベント var startStr = date + "T" + time + ":00"; var start = new Date(startStr); var end = new Date(start.getTime() + duration * 60 * 1000); event = calendar.createEvent( "[" + priority + "] " + title, start, end, { description: "タスクアプリから自動登録\n優先度: " + priority } ); } else if (date) { // 日付のみ → 終日イベント var parts = date.split("-"); var allDay = new Date( parseInt(parts[0]), parseInt(parts[1]) - 1, parseInt(parts[2]) ); event = calendar.createAllDayEvent( "[" + priority + "] " + title, allDay, { description: "タスクアプリから自動登録\n優先度: " + priority } ); } else { // 日付なし → 今日の終日イベント event = calendar.createAllDayEvent( "[" + priority + "] " + title, new Date(), { description: "タスクアプリから自動登録\n優先度: " + priority } ); } return ContentService .createTextOutput(JSON.stringify({ status: "ok", eventId: event.getId(), title: event.getTitle() })) .setMimeType(ContentService.MimeType.JSON) .setHeader ? addCorsHeaders(ContentService.createTextOutput(JSON.stringify({ status: "ok", eventId: event.getId(), title: event.getTitle() })).setMimeType(ContentService.MimeType.JSON)) : ContentService.createTextOutput(JSON.stringify({ status: "ok", eventId: event.getId(), title: event.getTitle() })).setMimeType(ContentService.MimeType.JSON); } catch(err) { return ContentService .createTextOutput(JSON.stringify({ status: "error", message: err.toString() })) .setMimeType(ContentService.MimeType.JSON); } } function doGet(e) { return ContentService .createTextOutput(JSON.stringify({ status: "ok", message: "タスクカレンダー連携 稼働中" })) .setMimeType(ContentService.MimeType.JSON); }
